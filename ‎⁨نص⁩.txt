        function processImageFile(file) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                reader.onload = (e) => {
                    const img = new Image();
                    img.onload = () => {
                        const canvas = document.createElement('canvas');
                        let maxDim = 600;
                        let w = img.width;
                        let h = img.height;

                        if (w > h) {
                            if (w > maxDim) {
                                h = Math.round((h * maxDim) / w);
                                w = maxDim;
                            }
                        } else {
                            if (h > maxDim) {
                                w = Math.round((w * maxDim) / h);
                                h = maxDim;
                            }
                        }

                        canvas.width = w;
                        canvas.height = h;
                        const ctx = canvas.getContext('2d');
                        ctx.drawImage(img, 0, 0, w, h);
                        resolve(canvas.toDataURL('image/jpeg', 0.85));
                    };
                    img.onerror = reject;
                    img.src = e.target.result;
                };
                reader.onerror = reject;
                reader.readAsDataURL(file);
            });
        }

        const DB_NAME = 'MomenSaraAudioDB';
        function initAudioDB() {
            return new Promise((resolve) => {
                const req = indexedDB.open(DB_NAME, 1);
                req.onupgradeneeded = (e) => {
                    const db = e.target.result;
                    if (!db.objectStoreNames.contains('files')) {
                        db.createObjectStore('files');
                    }
                };
                req.onsuccess = (e) => resolve(e.target.result);
                req.onerror = () => resolve(null);
            });
        }

                    const tx = db.transaction('files', 'readonly');
                    const req = tx.objectStore('files').get('userSong');
                    req.onsuccess = () => resolve(req.result || null);
                    req.onerror = () => resolve(null);
                });
            } catch (e) {
                return null;
            }
        }

            } catch (e) {}
        }

        function playPianoSynthNote(freq) {
            try {
                if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
                gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 1.2);
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start();
                osc.stop(audioCtx.currentTime + 1.2);
            } catch (e) {}
        }

            status.textContent = "اضغط للتشغيل 🎵";

        const canvas = document.getElementById('heartsCanvas');
        const ctx = canvas.getContext('2d');
        let hearts = [];

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        class Heart {
            constructor(x, y) {
                this.x = x || Math.random() * canvas.width;
                this.y = y || canvas.height + 20;
                this.size = Math.random() * 14 + 10;
                this.speedY = Math.random() * 2 + 1;
                this.speedX = (Math.random() - 0.5) * 1.5;
                this.opacity = Math.random() * 0.7 + 0.3;
                this.color = `hsl(${Math.random() * 30 + 340}, 90%, 65%)`;
            }

            update() {
                this.y -= this.speedY;
                this.x += this.speedX;
                this.opacity -= 0.004;
            }

            draw() {
                ctx.save();
                ctx.globalAlpha = Math.max(0, this.opacity);
                ctx.fillStyle = this.color;
                ctx.beginPath();
                const topCurveHeight = this.size * 0.3;
                ctx.moveTo(this.x, this.y + topCurveHeight);
                ctx.bezierCurveTo(this.x, this.y, this.x - this.size / 2, this.y, this.x - this.size / 2, this.y + topCurveHeight);
                ctx.bezierCurveTo(this.x - this.size / 2, this.y + (this.size + topCurveHeight) / 2, this.x, this.y + this.size, this.x, this.y + this.size);
                ctx.bezierCurveTo(this.x, this.y + (this.size + topCurveHeight) / 2, this.x + this.size / 2, this.y + (this.size + topCurveHeight) / 2, this.x + this.size / 2, this.y + topCurveHeight);
                ctx.bezierCurveTo(this.x + this.size / 2, this.y, this.x, this.y, this.x, this.y + topCurveHeight);
                ctx.closePath();
                ctx.fill();
                ctx.restore();
            }
        }

        function triggerHeartExplosion(x, y) {
            const startX = x || canvas.width / 2;
            const startY = y || canvas.height / 2;
            for (let i = 0; i < 25; i++) {
                const h = new Heart(startX, startY);
                h.speedY = (Math.random() - 0.5) * 5;
                h.speedX = (Math.random() - 0.5) * 5;
                hearts.push(h);
            }
        }

        window.addEventListener('click', (e) => {
            if (e.target.tagName !== 'BUTTON' && e.target.tagName !== 'INPUT') {
                hearts.push(new Heart(e.clientX, e.clientY));
            }
        });

        function animateHearts() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            if (Math.random() < 0.1) {
                hearts.push(new Heart());
            }

            for (let i = 0; i < hearts.length; i++) {
                hearts[i].update();
                hearts[i].draw();
                if (hearts[i].opacity <= 0 || hearts[i].y < -20) {
                    hearts.splice(i, 1);
                    i--;
                }
            }
            requestAnimationFrame(animateHearts);
        }

        window.onload = async function() {
            await loadConfig();
            runCounterEngine();
            setInterval(runCounterEngine, 1000);
            animateHearts();
        };
